Temel İşlemler

Özyineli ikili arama

int ikili_ara(const int a[], int sol, int sag, int x) {
    if (sol > sag) return -1;
    int orta = sol + (sag - sol) / 2;
    if (a[orta] == x) return orta;
    if (x < a[orta]) return ikili_ara(a, sol, orta - 1, x);
    return ikili_ara(a, orta + 1, sag, x);
}
Kod örnekleri C99/C11 uyumlu olacak biçimde küçük ve tek sorumluluklu tutulmuştur.
